//--- Input Parameters
input string  _1="========== MODE SETTINGS ==========";
input bool    EnableTPSL_Mode = true;           // Enable TP/SL Mode
input bool    EnableTrailing_Mode = false;     // Enable Trailing SL Mode
input bool    EnableMartingale_Mode = false;   // Enable Martingale TP Mode

input string  _2="========== TP/SL MODE ==========";
input double  TP_Pips = 10.0;                  // Take Profit (Pips)
input double  SL_Pips = 10.0;                  // Stop Loss (Pips)

input string  _3="========== TRAILING MODE ==========";
input double  Initial_SL_Pips = 10.0;          // Initial SL Distance (Pips)
input double  Start_Trailing_Pips = 2.0;       // Start Trailing at Profit (Pips)
input double  Trailing_Distance_Pips = 20.0;   // Trailing Distance (Pips)

input string  _4="========== MARTINGALE MODE ==========";
input double  TP_Dollar = 0.1;                 // TP Individual Position ($)
input double  TP_CutAll_Dollar = 0.1;          // TP Cut All Positions ($)

input string  _5="========== GENERAL SETTINGS ==========";
input double  LotSize = 0.01;                  // Lot Size
input int     MagicNumber = 12345;              // Magic Number
input bool    ShowEmergencyButton = true;      // Show Emergency Button

//--- Global Variables
datetime StartTime;

//+------------------------------------------------------------------+
//| Expert initialization function                                   |
//+------------------------------------------------------------------+
int OnInit()
{
   StartTime = TimeCurrent();
   
   if(ShowEmergencyButton)
      CreateEmergencyButton();
   
   Print("Position Manager EA Started - ", Symbol());
   Print("Mode 1 (TP/SL): ", EnableTPSL_Mode ? "ON" : "OFF");
   Print("Mode 2 (Trailing): ", EnableTrailing_Mode ? "ON" : "OFF");
   Print("Mode 3 (Martingale): ", EnableMartingale_Mode ? "ON" : "OFF");
   
   return(INIT_SUCCEEDED);
}

//+------------------------------------------------------------------+
//| Expert deinitialization function                                |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
   if(ShowEmergencyButton)
      RemoveEmergencyButton();
      
   Print("Position Manager EA Stopped");
}

//+------------------------------------------------------------------+
//| Expert tick function                                             |
//+------------------------------------------------------------------+
void OnTick()
{
   // Process Martingale Mode (Priority)
   if(EnableMartingale_Mode)
   {
      ProcessMartingaleMode();
   }
   
   // Process other modes only if not martingale or different magic numbers
   if(EnableTPSL_Mode || EnableTrailing_Mode)
   {
      for(int i = OrdersTotal() - 1; i >= 0; i--)
      {
         if(OrderSelect(i, SELECT_BY_POS, MODE_TRADES))
         {
            if(OrderSymbol() == Symbol())
            {
               // Skip if martingale mode is active and same magic number
               if(EnableMartingale_Mode && OrderMagicNumber() == MagicNumber)
                  continue;
                  
               if(EnableTPSL_Mode)
                  ManageTPSL();
                  
               if(EnableTrailing_Mode)
                  ManageTrailing();
            }
         }
      }
   }
}

//+------------------------------------------------------------------+
//| Handle Chart Events                                             |
//+------------------------------------------------------------------+
void OnChartEvent(const int id, const long& lparam, const double& dparam, const string& sparam)
{
   if(id == CHARTEVENT_OBJECT_CLICK && sparam == "EmergencyBtn")
   {
      EmergencyCut();
      ObjectSetInteger(0, "EmergencyBtn", OBJPROP_STATE, false);
   }
}

//+------------------------------------------------------------------+
//| Process Martingale Mode                                         |
//+------------------------------------------------------------------+
void ProcessMartingaleMode()
{
   // First check for Cut All condition
   double totalProfit = 0;
   int positionCount = 0;
   
   // Calculate total profit for magic number group
   for(int i = 0; i < OrdersTotal(); i++)
   {
      if(OrderSelect(i, SELECT_BY_POS, MODE_TRADES))
      {
         if(OrderSymbol() == Symbol() && OrderMagicNumber() == MagicNumber)
         {
            totalProfit += OrderProfit() + OrderSwap() + OrderCommission();
            positionCount++;
         }
      }
   }
   
   // Cut All if total profit reached
   if(positionCount > 0 && totalProfit >= TP_CutAll_Dollar)
   {
      CloseAllPositions();
      Print("Martingale Cut All Triggered - Total Profit: $", DoubleToString(totalProfit, 2));
      return;
   }
   
   // Check individual TP for each position
   for(int i = OrdersTotal() - 1; i >= 0; i--)
   {
      if(OrderSelect(i, SELECT_BY_POS, MODE_TRADES))
      {
         if(OrderSymbol() == Symbol() && OrderMagicNumber() == MagicNumber)
         {
            double positionProfit = OrderProfit() + OrderSwap() + OrderCommission();
            
            if(positionProfit >= TP_Dollar)
            {
               int ticket = OrderTicket();
               if(ClosePosition(ticket))
                  Print("Martingale Individual TP - Ticket: ", ticket, " Profit: $", DoubleToString(positionProfit, 2));
            }
         }
      }
   }
}

//+------------------------------------------------------------------+
//| Close All Positions                                            |
//+------------------------------------------------------------------+
void CloseAllPositions()
{
   for(int i = OrdersTotal() - 1; i >= 0; i--)
   {
      if(OrderSelect(i, SELECT_BY_POS, MODE_TRADES))
      {
         if(OrderSymbol() == Symbol() && OrderMagicNumber() == MagicNumber)
         {
            ClosePosition(OrderTicket());
         }
      }
   }
}

//+------------------------------------------------------------------+
//| Close Single Position                                           |
//+------------------------------------------------------------------+
bool ClosePosition(int ticket)
{
   if(OrderSelect(ticket, SELECT_BY_TICKET))
   {
      double closePrice;
      if(OrderType() == OP_BUY)
         closePrice = Bid;
      else if(OrderType() == OP_SELL)
         closePrice = Ask;
      else
         return false; // Pending orders
         
      bool result = OrderClose(ticket, OrderLots(), closePrice, 3, clrNONE);
      if(!result)
         Print("Error closing position: ", GetLastError());
      return result;
   }
   return false;
}

//+------------------------------------------------------------------+
//| Emergency Cut All Positions                                    |
//+------------------------------------------------------------------+
void EmergencyCut()
{
   int closedCount = 0;
   double totalClosed = 0;
   
   Print("=== EMERGENCY CUT ACTIVATED ===");
   
   for(int i = OrdersTotal() - 1; i >= 0; i--)
   {
      if(OrderSelect(i, SELECT_BY_POS, MODE_TRADES))
      {
         if(OrderSymbol() == Symbol() && OrderMagicNumber() == MagicNumber)
         {
            double profit = OrderProfit() + OrderSwap() + OrderCommission();
            int ticket = OrderTicket();
            
            if(ClosePosition(ticket))
            {
               closedCount++;
               totalClosed += profit;
               Print("Emergency Close - Ticket: ", ticket, " Profit: $", DoubleToString(profit, 2));
            }
         }
      }
   }
   
   Print("=== EMERGENCY CUT COMPLETED ===");
   Print("Positions Closed: ", closedCount);
   Print("Total P/L: $", DoubleToString(totalClosed, 2));
   
   Alert("EMERGENCY CUT - ", closedCount, " positions closed. P/L: $", DoubleToString(totalClosed, 2));
}

//+------------------------------------------------------------------+
//| Create Emergency Button                                         |
//+------------------------------------------------------------------+
void CreateEmergencyButton()
{
   ObjectCreate(0, "EmergencyBtn", OBJ_BUTTON, 0, 0, 0);
   ObjectSetInteger(0, "EmergencyBtn", OBJPROP_CORNER, CORNER_RIGHT_UPPER);
   ObjectSetInteger(0, "EmergencyBtn", OBJPROP_XDISTANCE, 10);
   ObjectSetInteger(0, "EmergencyBtn", OBJPROP_YDISTANCE, 10);
   ObjectSetInteger(0, "EmergencyBtn", OBJPROP_XSIZE, 100);
   ObjectSetInteger(0, "EmergencyBtn", OBJPROP_YSIZE, 30);
   ObjectSetString(0, "EmergencyBtn", OBJPROP_TEXT, "EMERGENCY");
   ObjectSetString(0, "EmergencyBtn", OBJPROP_FONT, "Arial Bold");
   ObjectSetInteger(0, "EmergencyBtn", OBJPROP_FONTSIZE, 9);
   ObjectSetInteger(0, "EmergencyBtn", OBJPROP_COLOR, clrWhite);
   ObjectSetInteger(0, "EmergencyBtn", OBJPROP_BGCOLOR, clrRed);
   ObjectSetInteger(0, "EmergencyBtn", OBJPROP_BORDER_COLOR, clrWhite);
}

//+------------------------------------------------------------------+
//| Remove Emergency Button                                         |
//+------------------------------------------------------------------+
void RemoveEmergencyButton()
{
   ObjectDelete(0, "EmergencyBtn");
}

//+------------------------------------------------------------------+
//| Manage TP/SL Mode                                               |
//+------------------------------------------------------------------+
void ManageTPSL()
{
   if(OrderType() == OP_BUY || OrderType() == OP_SELL)
   {
      double currentTP = OrderTakeProfit();
      double currentSL = OrderStopLoss();
      
      // Only set TP/SL if they are not already set
      if(currentTP == 0 || currentSL == 0)
      {
         double newTP = 0, newSL = 0;
         double pipValue = GetPipValue();
         
         if(OrderType() == OP_BUY)
         {
            newTP = OrderOpenPrice() + (TP_Pips * pipValue);
            newSL = OrderOpenPrice() - (SL_Pips * pipValue);
         }
         else if(OrderType() == OP_SELL)
         {
            newTP = OrderOpenPrice() - (TP_Pips * pipValue);
            newSL = OrderOpenPrice() + (SL_Pips * pipValue);
         }
         
         if(newTP > 0 && newSL > 0)
         {
            bool result = OrderModify(OrderTicket(), OrderOpenPrice(), newSL, newTP, 0, clrNONE);
            if(result)
               Print("TP/SL Set - Ticket: ", OrderTicket(), " TP: ", newTP, " SL: ", newSL);
         }
      }
   }
}

//+------------------------------------------------------------------+
//| Manage Trailing SL Mode                                         |
//+------------------------------------------------------------------+
void ManageTrailing()
{
   if(OrderType() == OP_BUY || OrderType() == OP_SELL)
   {
      double currentSL = OrderStopLoss();
      double pipValue = GetPipValue();
      double currentProfit = 0;
      
      // Calculate current profit in pips
      if(OrderType() == OP_BUY)
         currentProfit = (Bid - OrderOpenPrice()) / pipValue;
      else if(OrderType() == OP_SELL)
         currentProfit = (OrderOpenPrice() - Ask) / pipValue;
      
      // Set initial SL if not set
      if(currentSL == 0)
      {
         double newSL = 0;
         if(OrderType() == OP_BUY)
            newSL = OrderOpenPrice() - (Initial_SL_Pips * pipValue);
         else if(OrderType() == OP_SELL)
            newSL = OrderOpenPrice() + (Initial_SL_Pips * pipValue);
            
         if(newSL > 0)
         {
            bool result = OrderModify(OrderTicket(), OrderOpenPrice(), newSL, OrderTakeProfit(), 0, clrNONE);
            if(result)
               Print("Initial SL Set - Ticket: ", OrderTicket(), " SL: ", newSL);
         }
      }
      // Start trailing when profit reaches threshold
      else if(currentProfit >= Start_Trailing_Pips)
      {
         double newSL = 0;
         
         if(OrderType() == OP_BUY)
         {
            newSL = Bid - (Trailing_Distance_Pips * pipValue);
            if(newSL > currentSL && newSL > OrderOpenPrice())
            {
               bool result = OrderModify(OrderTicket(), OrderOpenPrice(), newSL, OrderTakeProfit(), 0, clrNONE);
               if(result)
                  Print("Trailing SL Updated - Ticket: ", OrderTicket(), " New SL: ", newSL);
            }
         }
         else if(OrderType() == OP_SELL)
         {
            newSL = Ask + (Trailing_Distance_Pips * pipValue);
            if((newSL < currentSL || currentSL == 0) && newSL < OrderOpenPrice())
            {
               bool result = OrderModify(OrderTicket(), OrderOpenPrice(), newSL, OrderTakeProfit(), 0, clrNONE);
               if(result)
                  Print("Trailing SL Updated - Ticket: ", OrderTicket(), " New SL: ", newSL);
            }
         }
      }
   }
}

//+------------------------------------------------------------------+
//| Get Pip Value                                                   |
//+------------------------------------------------------------------+
double GetPipValue()
{
   if(Digits == 5 || Digits == 3)
      return Point * 10;
   else
      return Point;
}
